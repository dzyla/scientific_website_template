{{/* Renders publications from data/publications.json as a numbered, year-grouped academic reference list. Expected fields per item: title, authors (";"-separated string), journal, year, link (optional), highlight (array of names), summary (optional) */}} {{- $pubs := site.Data.publications -}} {{- if not $pubs -}} {{- errorf "Could not find publications data in data/publications.json." -}} {{- else -}} {{- $sorted := sort $pubs "year" "desc" -}}
{{- $lastYear := "" -}} {{- range $sorted -}} {{- $p := . -}} {{- $currentYear := $p.year -}} {{- if ne $currentYear $lastYear -}} {{- if ne $lastYear "" -}}{{- end -}} {{- $count := len (where $sorted "year" $currentYear) -}}

{{ $currentYear }} {{ $count }} {{ cond (eq $count 1) "paper" "papers" }}

    {{- $lastYear = $currentYear -}} {{- end -}} {{- $authors := split ($p.authors | plainify) ";" -}} {{- $hl := $p.highlight | default slice -}} {{- $citeAuthors := replace $p.authors ";" "," -}} {{- $citeStr := printf "%s (%v). %s. %s." $citeAuthors $p.year ($p.title | plainify) ($p.journal | plainify) -}} {{- if $p.link -}}{{- $citeStr = printf "%s %s" $citeStr $p.link -}}{{- end -}} {{- $doi := $p.doi -}} {{- if and (not $doi) $p.link (hasPrefix $p.link "https://doi.org/") -}} {{- $doi = replaceRE "^https?://(dx\\.)?doi\\.org/" "" $p.link -}} {{- end -}}

  1. {{- if $p.link -}} {{ $p.title | plainify }} {{- else -}} {{ $p.title | plainify }} {{- end -}}

    {{- range $i, $name := $authors -}} {{- $trim := trim $name " " -}} {{- if in $hl $trim -}}{{ $trim }}{{ else }}{{ $trim }}{{ end -}} {{- if lt (add $i 1) (len $authors) -}}, {{ end -}} {{- end -}}

    {{- with $p.summary }}

    {{ . }}

    {{- end -}}
    {{ $p.journal | plainify }} · {{ $p.year }}{{ with $doi }} · {{ . }}{{ end }} {{- if $p.link -}} View paper {{- end -}}
    2. {{- end -}} {{- if ne $lastYear "" -}}
{{- end -}}
{{- end -}}
